            cout << "El auto ahora esta apagado" << endl;

#include<iostream>
#include<string>

using namespace std;


class Automotor { // Esto es encapsulamiento porque guardas atributos y metodos en un solo lugar 
    public:
        string modelo;
        int cilindros;
        float gasolina = 100;
        float velocidad = 0;
        bool encendido = false;

        void llenar_tanque() { // Nuevo metodo 
            gasolina = 100.0;
        }

        void encender(){
            encendido = true;
            cout << "El auto ahora esta encendido" << endl;

        }

        void apagar() {
            encendido = false;
            cout << "El auto ahora esta apagado" << endl;
        }

        virtual void consumo(int velocidad) { // Polimorfismo
            float consumo = .1*velocidad;
            gasolina -= consumo;
            cout << "Se consumio: " << consumo << "gasolina" << endl;
        }

        void acelerar(int cantidad) { // Abstraccion, abstraes el metodo a una llamada relativamente sencilla solo con el nombre
            // instruccion modificada 
            if(encendido && gasolina > 0) {
                velocidad += cantidad;
                cout << "El auto ahora esta avanzando con una velocidad de: " << velocidad << "km/h" << endl;
                consumo(velocidad);
            
            } else {
                cout << "el auto esta apagado, primero enciendelo" << endl;
            }
        }

};


class Trailer: public Automotor { // Herencia
    public: 
    void consumo(int velocidad){ // Polimorfismo
        float consumo = .3*velocidad;
        gasolina -= consumo;
        cout << "Se consumio: " << consumo << "% gasolina" << endl; 
    }
};


int main() {

    Trailer Mack;

    Mack.encender();

    Mack.acelerar(10);


}

#include<iostream>
#include<string>

#include<stdexcept>
using namespace std;

class CuentaBancaria {
    private:
        string username;
        double saldo = 0;
    public: 
        CuentaBancaria(string name): username{name}  {} // solo funcuona cuando se tienen atributos en public y private
        
        double devolver_saldo() const {return saldo;} // accessor 

        void depositar(double cantidad) { // mutator
            if(cantidad <= 0) { // invariante de clase
                throw invalid_argument("Monto ingresado para desposito invalido");
            }

            saldo += cantidad;
        }

        void leer_saldo() {
            cout << "Saldo del usuario " << username << " es: " << saldo << endl;
        }
};      



int main() {

    CuentaBancaria rivas{"David Rivas"}; // solo funciona cuando todos los atributos estan en public, cuando no toca constructor
    cout << "probando accessor: " << rivas.devolver_saldo() << endl;

    rivas.depositar(100);
    cout << "probando accessor 2: " << rivas.devolver_saldo() << endl;


}
