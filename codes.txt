------------ Cadena if else

#include <iostream>

int main() {
    int velocidad = 75; 

    if (velocidad > 120) {
        std::cout << "Peligro: Exceso de velocidad extremo.\n";
    } 
    else if (velocidad > 80) {
        std::cout << "Advertencia: Vas a alta velocidad.\n";
    } 
    else if (velocidad > 40) {
        std::cout << "Velocidad moderada y adecuada.\n";
    } 
    else {
        std::cout << "Vehiculo detenido o demasiado lento.\n";
    }

}

------------ C++

#include <iostream>

int main() {
    char calificacion = 'B';

    switch (calificacion) {
        case 'A':
            std::cout << "¡Excelente trabajo!" << std::endl;
            break; 
            
        case 'B':
            std::cout << "Buen rendimiento." << std::endl;
            break; 
            
        case 'C':
            std::cout << "Aprobado, pero puedes mejorar." << std::endl;
            break; 
            
        default: 
            std::cout << "Calificación no válida." << std::endl;
            break; 
    }

}

------------ Cadena if else

fn main() {
    let calificacion = 'B';

    match calificacion {
        'A' => println!("¡Excelente trabajo!"),
        'B' => println!("Buen rendimiento."),
        'C' => println!("Aprobado, pero puedes mejorar."),
        _ => println!("Calificación no válida."), 
    }
}

for con vectores e index ------------

#include <iostream>
#include <vector> 

int main() {
    std::vector<int> calificaciones = {85, 92, 78, 90, 88};

    for (size_t i = 0; i < calificaciones.size(); ++i) {
        std::cout << "Indice [" << i << "]: " << calificaciones[i] << "\n";
    }
}

------------ while

#include <iostream>

int main() {
    int contador = 1; 

    while (contador <= 3) {
        std::cout << "Iteracion #" << contador << "\n";
        
        contador++; 
        contador = contador + 1;
        contador += 1;
        contador = contador++; // incorrecto 
        contador = ++contador; // correcto

    }

}

------------ do while


#include <iostream>

int main() {
    int numero; 

    do {
        std::cout << "Introduce un numero positivo de un digito (1 - 9): ";
        std::cin >> numero; 
        
    } while (numero < 1 || numero > 9); 

    std::cout << "¡Excelente! El numero " << numero << " es valido.\n";
}
