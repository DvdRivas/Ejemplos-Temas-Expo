------------ Cadena if else

#include <iostream>

int main() {
    int velocidad = 75; 

    if (velocidad > 120) {
        std::cout << "Peligro: Exceso de velocidad extremo.\n";
    } 
    else if (velocidad > 80) {
        std::cout << "Advertencia: Vas a alta velocidad.\n";
    } 
    else if (velocidad > 40) {
        std::cout << "Velocidad moderada y adecuada.\n";
    } 
    else {
        std::cout << "Vehiculo detenido o demasiado lento.\n";
    }

}

------------ C++

#include <iostream>

int main() {
    char calificacion = 'B';

    switch (calificacion) {
        case 'A':
            std::cout << "¡Excelente trabajo!" << std::endl;
            break; 
            
        case 'B':
            std::cout << "Buen rendimiento." << std::endl;
            break; 
            
        case 'C':
            std::cout << "Aprobado, pero puedes mejorar." << std::endl;
            break; 
            
        default: 
            std::cout << "Calificación no válida." << std::endl;
            break; 
    }

}

------------ Cadena if else

fn main() {
    let calificacion = 'B';

    match calificacion {
        'A' => println!("¡Excelente trabajo!"),
        'B' => println!("Buen rendimiento."),
        'C' => println!("Aprobado, pero puedes mejorar."),
        _ => println!("Calificación no válida."), 
    }
}

------------ while

#include <iostream>

int main() {
    int contador = 1; 

    while (contador <= 3) {
        std::cout << "Iteracion #" << contador << "\n";
        
        contador++; 
        contador = contador + 1;
        contador += 1;
        contador = contador++; // incorrecto 
        contador = ++contador; // correcto

    }

}

#include <iostream>

int main() {
    int numero; 

    do {
        std::cout << "Introduce un numero positivo de un digito (1 - 9): ";
        std::cin >> numero;

------------- Ciclos anidados, actividad, crear un contador que vaya de 1 a 100 usando dos for

int main() {
    int valor;

    for (int j = 0; j <= 9; j++) {
        for (int k = 0; k <= 9; k++) {
            valor  =  (j * 10) + k + 1;
            std::cout << valor << " ";
        }
    }
}
